50template <class T>
51bool findAugmentingPath(Graph<T> *g, Vertex<T> *s, Vertex<T> *t) {
52 for(auto v : g->getVertexSet()) {
53 v->setVisited(false);
54 }
55 std::queue<Vertex<T>*> q;
56 s->setVisited(true);
57 q.push(s);
58
59 while (!q.empty()) {
60 auto u = q.front(); q.pop();
61
62 for (auto e : u->getAdj()) {
63 auto v = e->getDest();
64 double residual = e->getWeight() - e->getFlow();
65 testAndVisit(q, e, v, residual);
66 }
67
68 for (auto e : u->getIncoming()) {
69 auto v = e->getOrig();
70 double residual = e->getFlow();
71 testAndVisit(q, e, v, residual);
72 }
73 }
74 return t->isVisited();
75}

89template <class T>
90double findMinResidualAlongPath(Vertex<T> *s, Vertex<T> *t) {
91 double f = INF;
92 Vertex<T>* curr = t;
93 while (curr != s) {
94 auto e = curr->getPath();
95 // Verifica se a aresta é direta ou inversa para calcular o residual corretamente
96 double residual = (e->getDest() == curr) ? (e->getWeight() - e->getFlow()) : e->getFlow();
97 f = std::min(f, residual);
98 curr = (e->getDest() == curr) ? e->getOrig() : e->getDest();
99 }
100 return f;
101}

115template <class T>
116void augmentFlowAlongPath(Vertex<T> *s, Vertex<T> *t, double f) {
117 Vertex<T>* cur = t;
118 while (cur != s) {
119 auto e = cur->getPath();
120 if (e->getDest() == cur) {
121 e->setFlow(e->getFlow() + f);
122 cur = e->getOrig();
123 } else {
124 e->setFlow(e->getFlow() - f);
125 cur = e->getDest();
126 }
127 }
128}

144template <class T>
145void edmondsKarp(Graph<T> *g, T sourceInfo, T targetInfo) {
146 Vertex<T>* s = g->findVertex(sourceInfo);
147 Vertex<T>* t = g->findVertex(targetInfo);
148
149 // SEGURANÇA: Se a fonte ou o poço não existirem, paramos aqui
150 if (s == nullptr || t == nullptr) {
151 return;
152 }
153
154 // Inicializa o fluxo de todas as arestas a 0 antes de começar
155 for (auto v : g->getVertexSet()) {
156 for (auto e : v->getAdj()) {
157 e->setFlow(0);
158 }
159 }
160
161 // Enquanto houver caminho aumentante, o fluxo continua a ser somado
162 while (findAugmentingPath(g, s, t)) {
163 double f = findMinResidualAlongPath(s, t);
164 augmentFlowAlongPath(s, t, f);
165 }
166}
